\noindent\textbf{Leyenda de Roles RACI:}
\vspace{0.2cm}
\begin{itemize}[itemsep=0.1cm]
    \item \colorbox{colR}{\makebox[0.5cm]{\textbf{R}}} \textbf{Responsable (Responsible):} Quien ejecuta la tarea y desarrolla el trabajo.
    \item \colorbox{colA}{\makebox[0.5cm]{\textbf{A}}} \textbf{Aprobador (Accountable):} Quien revisa y aprueba el resultado. Es el responsable final.
    \item \colorbox{colC}{\makebox[0.5cm]{\textbf{C}}} \textbf{Consultado (Consulted):} Brinda información, apoyo técnico o asesoría durante la ejecución.
    \item \colorbox{colI}{\makebox[0.5cm]{\textbf{I}}} \textbf{Informado (Informed):} Recibe notificaciones sobre los avances o la finalización del entregable.
\end{itemize}
